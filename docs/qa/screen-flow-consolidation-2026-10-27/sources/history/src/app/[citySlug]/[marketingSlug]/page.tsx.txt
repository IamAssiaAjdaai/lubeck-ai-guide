import type { Metadata } from "next";
import Link from "next/link";
import { notFound } from "next/navigation";
import { connection } from "next/server";
import { CitywalkLogo } from "@/components/brand/CitywalkLogo";
import MarketingLandingTracker from "@/components/marketing/MarketingLandingTracker";
import { appendMarketingAttributionToPath, readMarketingAttribution } from "@/lib/marketingAttribution";
import { getPublishedMarketingPageByPath } from "@/lib/marketing/publicRepository.server";

type Props=Readonly<{params:Promise<{citySlug:string;marketingSlug:string}>;searchParams?:Promise<Readonly<Record<string,string|string[]|undefined>>>}>;
async function load(citySlug:string,marketingSlug:string){await connection();return getPublishedMarketingPageByPath(`/${citySlug}/${marketingSlug}`);}
export async function generateMetadata({params}:Props):Promise<Metadata>{const {citySlug,marketingSlug}=await params;const page=await load(citySlug,marketingSlug);if(!page||page.city.slug!==citySlug)return {};const title=page.seoTitle??page.title;const description=page.seoDescription;const canonical=`https://getcitywalk.com${page.path}`;return {title,...(description?{description}:{}),alternates:{canonical},openGraph:{title,...(description?{description}:{}),url:canonical,siteName:"CITYWALK",type:page.kind==="blog"?"article":"website"}};}
export default async function MarketingPage({params,searchParams}:Props){const {citySlug,marketingSlug}=await params;const page=await load(citySlug,marketingSlug);if(!page||page.city.slug!==citySlug)notFound();const attribution=readMarketingAttribution((await searchParams)??{});const plannerHref=appendMarketingAttributionToPath(`/en/${citySlug}`,attribution);const paragraphs=page.body.split(/\n\s*\n/).map(x=>x.trim()).filter(Boolean);return <main className="min-h-screen bg-white text-text-primary"><div className="mx-auto flex min-h-screen w-full max-w-5xl flex-col px-5 py-6 sm:px-8 sm:py-8"><header className="flex items-center justify-between"><Link href="/" aria-label="CITYWALK home"><CitywalkLogo className="inline-flex items-center gap-2" markClassName="h-7 w-12 text-accent" wordmarkClassName="text-sm font-bold tracking-[0.16em]"/></Link><span className="text-xs font-semibold uppercase tracking-[0.16em] text-text-muted">{page.city.name}</span></header><section className="flex flex-1 items-center py-14 sm:py-20"><div className="max-w-3xl"><p className="eyebrow">{page.kind==="blog"?"CITYWALK Journal":"Explore more. Plan less."}</p><h1 className="mt-4 text-[clamp(2.5rem,8vw,5.5rem)] font-bold leading-[0.96] tracking-[-0.055em]">{page.title}</h1><div className="mt-7 max-w-2xl space-y-4 text-lg leading-8 text-text-secondary">{paragraphs.map((p,i)=><p key={i}>{p}</p>)}</div><div className="mt-8"><MarketingLandingTracker city={page.city.slug} landingVariant={page.slug} campaignSlug={page.campaign?.slug} attribution={attribution} ctaHref={page.cta?plannerHref:undefined} ctaLabel={page.cta}/></div></div></section></div></main>}
