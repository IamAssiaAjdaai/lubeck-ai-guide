import Link from "next/link";
import { getBrandWorkspace } from "@/lib/marketing/brandContext/service.server";
import { correctionBrief } from "@/lib/marketing/brandContext/ui";
import { getMarketingEditorOptions } from "@/lib/admin/marketing/service.server";
import { AdminAuthorizationError } from "@/lib/admin/authorization.server";
import { BrandContextPersistenceError } from "@/lib/marketing/brandContext/service.server";
import { AdminPageHeader, EmptyContent } from "@/components/admin/AdminContentUi";
import { BrandBriefForm } from "@/components/admin/BrandBriefForm";
import { BrandReviewPanel, BrandCampaignAssociation } from "@/components/admin/BrandReviewPanel";
import { confirmBrandAction, correctBrandAction, associateBrandCampaignAction } from "../../actions";
export const metadata = { title: "Markenkontext | CITYWALK Admin", robots: { index: false, follow: false } };
export const dynamic = "force-dynamic";
export default async function BrandDetailPage({ params }: { params: Promise<{ projectId: string; contextId: string }> }) {
  const ids = await params;
  let workspace;
  try { workspace = await getBrandWorkspace(ids); }
  catch (error) { if (error instanceof AdminAuthorizationError || error instanceof BrandContextPersistenceError && error.code === "NOT_FOUND") return <EmptyContent>Projekt nicht verfügbar oder keine Berechtigung für alle zugeordneten Städte. <Link href="/admin/marketing/brands" className="text-primary">Zur Markenübersicht</Link></EmptyContent>; throw error; }
  const { context: c, scopes, campaigns, history } = workspace;
  const options = workspace.canManage ? await getMarketingEditorOptions() : null;
  const proof = c.facts.filter(f => f.confirmation).length;
  return <section className="space-y-6 break-words"><Link href="/admin/marketing/brands" className="text-sm font-semibold text-primary">← Marken & Projekte</Link><AdminPageHeader eyebrow={`Marke · Revision ${c.revision}`} title={c.identity.name} description={c.identity.summary ?? "Beschreibung noch offen."} />
    <p className="rounded-xl border border-border bg-blue-50 p-4 text-sm">{proof} von {c.facts.length} Aussagen menschlich bestätigt. Markenangaben sind keine unabhängig verifizierten CITYWALK-Ortsinformationen. Keine automatische Generierung oder Veröffentlichung.</p>
    <div className="grid gap-4 sm:grid-cols-2"><article className="surface-card min-w-0 p-5"><h2 className="text-xl font-bold">Geschäft & Ausrichtung</h2><p className="mt-3">{c.identity.businessType ?? "Unternehmensart offen"}</p><h3 className="mt-4 font-bold">Angebote</h3><ul>{c.offerings.map(o => <li key={o.id} className="mt-2"><strong>{o.name}</strong> · {o.description}</li>)}</ul><h3 className="mt-4 font-bold">Zielgruppen</h3><ul>{c.targetAudiences.map(a => <li key={a.id}>{a.description}</li>)}</ul><h3 className="mt-4 font-bold">Marketingziele</h3><ul>{c.marketingGoals.map(g => <li key={g.id}>{g.description}{g.cta ? ` · CTA: ${g.cta}` : ""}</li>)}</ul><p className="mt-4">Tonalität: {c.tone.voice.join(", ") || "offen"}</p><p>Zu vermeiden: {c.tone.avoid.join(", ") || "keine Angaben"}</p><p>Unzulässige Aussagen: {c.tone.forbiddenClaims.join(", ") || "keine Angaben"}</p><p>Märkte: {c.markets.map(m => m.name).join(", ") || "offen"}</p><p>Sprachen: {c.languages.join(", ") || "offen"}</p>{c.identity.website ? <a className="text-primary underline" href={c.identity.website} target="_blank" rel="noopener noreferrer">Unternehmenswebsite ↗</a> : null}</article>
    <article className="surface-card min-w-0 p-5"><h2 className="text-xl font-bold">Kampagnen & Stadtzugriffe</h2><p className="mt-3">Dauerhafte Stadtzugriffe: {scopes.map(s => s.name).join(", ")}</p><ul className="mt-3">{campaigns.map(k => <li key={k.id}>{k.name} · {scopes.find(s => s.cityId === k.cityId)?.name}</li>)}</ul>{!campaigns.length ? <p>Keine Kampagne zugeordnet. Die Marke und ihre bisherigen Stadtzugriffe bleiben erhalten.</p> : null}<h3 className="mt-5 font-bold">Business Brief</h3><p className="mt-2 whitespace-pre-wrap text-sm">{c.rawText}</p></article></div>
    <section className="surface-card p-5"><h2 className="text-xl font-bold">Fakten & Quellen</h2><p className="mt-2 text-sm">Quellen wurden manuell angegeben; Inhalte werden weder abgerufen noch unabhängig geprüft.</p>{c.facts.map(f => <article key={f.id} className="mt-5 border-t border-border pt-4"><p className="text-xs font-bold text-primary">{f.confirmation ? "Menschlich bestätigt" : "Vorgeschlagen · unbestätigt"}</p><p className="mt-2 font-semibold">{f.statement}</p><p className="mt-2 text-sm">Grundlage: {f.basis === "source_backed" ? "Quellenreferenz, nicht unabhängig verifiziert" : "Eigene Angabe"}</p><ul className="mt-2 text-sm">{(f.confirmation?.provenance ?? f.provenanceIds.map(id => c.provenance.find(p => p.id === id)!)).map(p => <li key={p.id}>{p.label} · {p.kind}{p.locator ? <> · <a className="text-primary underline" href={p.locator} target="_blank" rel="noopener noreferrer">Quelle öffnen ↗</a></> : null}{p.observedAt ? ` · ${new Date(p.observedAt).toLocaleString("de-DE", { timeZone: "Europe/Berlin" })}` : " · Datum nicht angegeben"}</li>)}</ul>{f.confirmation ? <p className="mt-2 break-all text-xs">Prüfkennung {f.confirmation.actorId} · {new Date(f.confirmation.confirmedAt).toLocaleString("de-DE", { timeZone: "Europe/Berlin" })}</p> : null}</article>)}{!c.facts.length ? <p className="mt-4">Noch keine Fakten vorgeschlagen.</p> : null}<details className="mt-4"><summary className="cursor-pointer font-semibold">Alle Quellenreferenzen</summary><ul className="mt-2 text-sm">{c.provenance.map(p => <li key={p.id} className="mt-2">{p.label} · {p.kind}{p.locator ? <> · <a className="text-primary underline" rel="noopener noreferrer" target="_blank" href={p.locator}>Quelle öffnen ↗</a></> : null}</li>)}</ul></details></section>
    {workspace.canConfirm && c.facts.some(f => !f.confirmation) ? <BrandReviewPanel key={c.revision} context={{ projectId: c.projectId, contextId: c.contextId, revision: c.revision, facts: c.facts }} action={confirmBrandAction} /> : <p className="text-sm">{workspace.canConfirm ? "Keine unbestätigten Fakten vorhanden." : "Faktenbestätigung erfordert Prüfberechtigung in allen Stadtzugriffen."}</p>}
    <div className="grid gap-4 sm:grid-cols-2"><section className="surface-card p-5"><h2 className="text-xl font-bold">Hypothesen · nicht bestätigt</h2><ul className="mt-3">{c.hypotheses.map(h => <li key={h.id} className="mt-3">{h.statement}{h.rationale ? <p className="text-sm text-text-secondary">{h.rationale}</p> : null}</li>)}</ul>{!c.hypotheses.length ? <p>Keine Hypothesen erfasst.</p> : null}</section><section className="surface-card p-5"><h2 className="text-xl font-bold">Offene Informationen</h2><ul className="mt-3">{c.missingInformation.map(m => <li key={m.id} className="mt-2">{m.question}</li>)}</ul></section></div>
    {workspace.canManage ? <details className="surface-card p-5"><summary className="cursor-pointer font-bold">Markenkontext korrigieren</summary><BrandBriefForm key={c.revision} initial={correctionBrief(c)} action={correctBrandAction} target={{ ...ids, expectedRevision: c.revision }} /></details> : null}
    {workspace.canManage && options ? <BrandCampaignAssociation {...ids} campaigns={options.campaigns.filter(k => !campaigns.some(campaign => campaign.id === k.id))} action={associateBrandCampaignAction} /> : null}
    <section className="surface-card p-5"><h2 className="text-xl font-bold">Revisionsverlauf</h2><p className="mt-2 text-sm">Prüfkennungen sind pseudonyme Zuordnungen; hier werden keine Kontonamen oder E-Mails aufgelöst.</p>{workspace.historyTruncated ? <p className="mt-2 text-sm">Die letzten 20 Revisionen werden angezeigt. Ältere Revisionen bleiben gespeichert.</p> : null}<ol className="mt-4 space-y-4">{history.map(h => <li key={h.revision} className="border-t border-border pt-3 text-sm"><p className="font-bold">Revision {h.revision} · {h.event === "created" ? "Angelegt" : h.event === "corrected" ? "Korrigiert · Bestätigungen zurückgesetzt" : "Fakten menschlich bestätigt"}</p><p>{new Date(h.occurredAt).toLocaleString("de-DE", { timeZone: "Europe/Berlin" })}</p><p className="break-all text-xs">Prüfkennung {h.actorId}</p><ul>{h.facts.map((statement, i) => <li key={i} className="mt-2">{statement}</li>)}</ul></li>)}</ol></section>
  </section>;
}
