export default function PlaceScreen() {
  const { citySlug, placeSlug } = useLocalSearchParams();
  // A different place must not inherit local audio, completion or scroll state.
  return <PlaceContent key={`${citySlug}:${placeSlug}`} />;
}

function formatVerifiedDate(locale: string, value: string) {
  const timestamp = Date.parse(value);
  if (Number.isNaN(timestamp)) return value;
  try {
    return new Intl.DateTimeFormat(locale, { dateStyle: "medium" }).format(timestamp);
  } catch {
    return new Intl.DateTimeFormat("en", { dateStyle: "medium" }).format(timestamp);
  }
}

function PlaceContent() {
  const params = useLocalSearchParams<{
    citySlug?: string | string[];
    placeSlug?: string | string[];
    focus?: string;
    tripVersion?: string | string[];
    tripId?: string | string[];
    tripStops?: string | string[];
    tripIndex?: string | string[];
    tripSource?: string | string[];
  }>();
  const scroll = useRef<ScrollView>(null);
  const contentOffset = useRef<number | undefined>(undefined);
  const sections = useRef<{ overview?: number; story?: number; audio?: number }>({});
  const focusedSection = useRef<string | undefined>(undefined);
  const focusSection = useCallback(() => {
    const focus = params.focus;
    if (focus !== "audio" && focus !== "story") return;
    const offset = sections.current[focus];
    if (
      offset === undefined ||
      contentOffset.current === undefined ||
      !scroll.current ||
      focusedSection.current === focus
    )
      return;
    scroll.current?.scrollTo({
      y: contentOffset.current + offset,
      animated: true,
    });
    focusedSection.current = focus;
  }, [params.focus]);
  useFocusEffect(
    useCallback(() => {
      focusSection();
      return () => {
        focusedSection.current = undefined;
      };
    }, [focusSection]),
  );
  const identity = parsePlaceRouteIdentity(params.citySlug, params.placeSlug);
  const trip = parseMobileTripContext(params);
  const { direction, locale, messages } = useNativeLocale();
  const t = walkCopy(locale);
  const copy = placeDetailCopy[locale];
  const { expanded } = useResponsiveTextLayout();
  const { fontScale } = useWindowDimensions();
  const premium = usePremiumAccess();
  const [saved, setSaved] = useState(false),
    [saveMessage, setSaveMessage] = useState("");
  useFocusEffect(
    useCallback(() => {
      let alive = true;
      void loadSavedPlaces()
        .then((items) => {
          if (alive)
            setSaved(
              items.some(
                (p) =>
                  p.citySlug === params.citySlug && p.slug === params.placeSlug,
              ),
            );
        })
        .catch(() => {
          if (alive) setSaveMessage(messages.tripSaveFailed);
        });
      return () => {
        alive = false;
      };
    }, [
      params.citySlug,
      params.placeSlug,
      messages.tripSaveFailed,
      setSaveMessage,
      setSaved,
    ]),
  );
  const [savingPlace, setSavingPlace] = useState(false);
  const [tripCompleted, setTripCompleted] = useState(false);
  const [sourcesExpanded, setSourcesExpanded] = useState(false);
  const placeState = usePublicPlace(
    identity?.citySlug ?? "invalid",
    identity?.placeSlug ?? "invalid",
    locale,
  );
  const guideState = useGuideEligibility(
    identity?.citySlug ?? "invalid",
    identity?.placeSlug ?? "invalid",
  );

  if (!identity)
    return (
      <Screen>
        <StatusMessage>{messages.unavailable}</StatusMessage>
      </Screen>
    );
  if (placeState.status === "loading")
    return (
      <Screen>
        <CitywalkLoading variant="place" />
      </Screen>
    );
  if (placeState.status === "error")
    return (
      <Screen>
        <ContentRecovery
          retry={placeState.retry}
          citySlug={identity.citySlug}
        />
      </Screen>
    );

  const place = placeState.data.place;
  if (
    placeState.data.city.slug !== identity.citySlug ||
    place.slug !== identity.placeSlug
  ) {
    return (
      <Screen>
        <StatusMessage>{messages.unavailable}</StatusMessage>
      </Screen>
    );
  }
  const directionsUrl =
    Platform.OS === "ios"
      ? `https://maps.apple.com/?daddr=${place.coordinates.lat},${place.coordinates.lng}&dirflg=w`
      : `https://www.google.com/maps/dir/?api=1&destination=${place.coordinates.lat},${place.coordinates.lng}&travelmode=walking`;
  const contentTextStyle = nativeContentTextStyle(
    locale,
    place.resolvedLocale,
  );
  // Reuse the exact city/place identity and approved media policy from Explore.
  // Unverified legacy paths never acquire an implied license in this screen.
  const photo = explorePhotography(identity.citySlug, place);
  const centralCredit = photo?.credit ? photoCreditEntries.find(entry => entry.source === photo.credit?.source) : undefined;
  const photoUri = photo?.uri ? selectImageUrl(selectPrimaryImageMedia(place.media), undefined, undefined, "detail") ?? photo.uri : undefined;
  const audio = selectExactLocaleAudio(place.media, locale);
  const hasStory = Boolean(place.content.story?.trim());
  const verifiedSources = placeState.data.verifiedSources ?? [];
  const latestVerifiedAt = verifiedSources.reduce<string | undefined>(
    (latest, source) =>
      !latest || Date.parse(source.verifiedAt) > Date.parse(latest)
        ? source.verifiedAt
        : latest,
    undefined,
  );
  const previousStop = trip ? adjacentMobileTripParams(trip, -1) : undefined;
  const nextStop = trip ? adjacentMobileTripParams(trip, 1) : undefined;

  return (
    <Screen scrollViewRef={scroll} navigation={false}>
      <Stack.Screen
        options={{
          title: nativePlaceName(
            identity.citySlug,
            place.slug,
            place.content.name,
            locale,
          ),
        }}
      />
      {saveMessage ? <StatusMessage>{saveMessage}</StatusMessage> : null}
      {trip ? (
        <TripProgress
          current={trip.currentStopIndex + 1}
          label={messages.stopProgress
            .replace("{current}", String(trip.currentStopIndex + 1))
            .replace("{total}", String(trip.stopSlugs.length))}
          total={trip.stopSlugs.length}
        />
      ) : null}
      <MotionView
        duration={motion.screen}
        style={styles.content}
        onLayout={(event) => {
          contentOffset.current = event.nativeEvent.layout.y;
          focusSection();
        }}
      >
        <View style={styles.heroFrame}>
          {photo ? <Image
            source={photoUri ? { uri: citywalkApi.resolveUrl(photoUri) } : photo?.source}
            cachePolicy="memory-disk" contentFit="cover" style={styles.hero}
            accessibilityLabel={nativePlaceName(identity.citySlug, place.slug, place.content.name, locale)}
            transition={motion.component}
          /> : <View testID="place-photo-fallback" style={[styles.hero, styles.photoFallback]}>
            <View accessibilityElementsHidden importantForAccessibility="no-hide-descendants" style={styles.placeholderMedallion}>
              <NativeIcon ios={place.category === "eat" ? "fork.knife" : place.category === "fun" ? "sparkles" : "building.2"}
                android={place.category === "eat" ? "restaurant" : place.category === "fun" ? "auto_awesome" : "location_city"} size={42} color={colors.primary} />
            </View>
            <AppText variant="metadata" style={styles.muted}>{copy.noPhoto}</AppText>
          </View>}
        </View>
        <View style={[styles.introduction, nativeTextBlock(direction)]}>
          <View style={[styles.titleRow, nativeRowStyle(direction), expanded && styles.stacked]}>
            <AppText accessibilityRole="header" variant="screenTitle" style={[styles.placeTitle, fontScale > 1.5 && styles.accessibleTitle]}>
              {nativePlaceName(
                identity.citySlug,
                place.slug,
                place.content.name,
                locale,
              )}
            </AppText>
            <PressableSurface
              accessibilityLabel={translate(
                locale,
                saved ? "common.saved" : "common.save",
              )}
              accessibilityRole="button"
              accessibilityState={{
                selected: saved,
                busy: savingPlace,
                disabled: savingPlace,
              }}
              disabled={savingPlace}
              style={styles.saveAction}
              onPress={() => {
                if (savingPlace) return;
                setSavingPlace(true);
                void toggleSavedPlace({
                  citySlug: identity.citySlug,
                  slug: place.slug,
                  name: place.content.name,
                })
                  .then(setSaved)
                  .catch(() => setSaveMessage(messages.tripSaveFailed))
                  .finally(() => setSavingPlace(false));
              }}
            >
              <NativeIcon
                ios={saved ? "heart.fill" : "heart"}
                android={saved ? "favorite" : "favorite_border"}
                color={colors.primary}
                size={21}
              />
              <AppText style={styles.saveLabel}>{translate(locale, saved ? "common.saved" : "common.save")}</AppText>
            </PressableSurface>
          </View>
          <View style={[styles.metadataRow, nativeRowStyle(direction)]}>
            <AppText variant="metadata" style={styles.eyebrow}>{discoveryCopy(locale)[place.category]}</AppText>
            <View style={[styles.duration, nativeRowStyle(direction)]}>
              <NativeIcon ios="clock" android="schedule" size={16} color={colors.primary} />
              <AppText variant="metadata" style={styles.muted}>{place.durationMinutes} {messages.visitMinutes}</AppText>
            </View>
            {centralCredit ? <Link href={{ pathname: "/photo-credits", params: { image: centralCredit.id } }} asChild>
              <PressableSurface accessibilityRole="link" accessibilityLabel={`${photoCreditsCopy(locale).title}: ${centralCredit.title}`} style={styles.photoCreditLink}>
                <AppText style={styles.photoCreditLabel}>{photoCreditsCopy(locale).title}</AppText>
              </PressableSurface>
            </Link> : null}
            {place.pricing && place.pricing !== "unknown" ? <AppText variant="metadata" style={styles.muted}>{copy[place.pricing]}</AppText> : null}
          </View>
          {direction === "rtl" && place.didFallback ? (
            <AppText variant="caption">{messages.fallbackContent}</AppText>
          ) : null}
        </View>
        {!centralCredit && (photo?.credit || photo?.attribution) ? <MediaAttribution attribution={photo.credit ? {
          text: `${photo.credit.credit} · Wikimedia Commons ${photo.credit.source} · ${photo.credit.licenseName} ${photo.credit.license}`,
        } : photo.attribution} /> : null}
        <View style={[styles.quickActions, nativeRowStyle(direction), expanded && styles.stacked]}>

          <View style={styles.quickAction}>
            <WalkMembershipControl
              citySlug={identity.citySlug}
              placeSlug={place.slug}
              dense
            />
          </View>
          <PrimaryButton
            wrapLabel
            label={translate(locale, "place.directions")}
            tone="secondary"
            style={styles.directionAction}
            leadingIcon={
              <NativeIcon
                ios="location"
                android="directions"
                color={colors.primary}
                size={19}
              />
            }
            onPress={() => {
              void Linking.openURL(directionsUrl).catch(() =>
                setSaveMessage(messages.unavailable),
              );
            }}
          />
        </View>
        <View style={[styles.sectionNavigation, nativeRowStyle(direction)]}>
          {([{ key: "overview" as const, label: copy.overview }, ...(hasStory ? [{ key: "story" as const, label: messages.story }] : []), ...(audio ? [{ key: "audio" as const, label: messages.audioGuide }] : [])]).map(section => <PressableSurface
            key={section.key} accessibilityRole="button" accessibilityLabel={section.label}
            onPress={() => { const offset = sections.current[section.key]; if (offset !== undefined && contentOffset.current !== undefined) scroll.current?.scrollTo({ y: contentOffset.current + offset, animated: true }); }} style={styles.sectionLink}>
            <AppText variant="label" style={styles.sectionLinkText}>{section.label}</AppText>
          </PressableSurface>)}
        </View>
        <View testID="place-overview-section" style={styles.section} onLayout={event => { sections.current.overview = event.nativeEvent.layout.y; }}>
          <AppText accessibilityRole="header" style={styles.sectionTitle}>{copy.overview}</AppText>
          <AppText style={[contentTextStyle, styles.description]}>
            {place.content.description ?? place.content.shortDescription}
          </AppText>
        </View>
        {place.content.visitNote ? (
          <View style={styles.note}>
            <View style={styles.noteHeading}>
              <NativeIcon
                ios="info.circle"
                android="info"
                color={colors.primary}
                size={19}
              />
              <AppText variant="label" style={styles.noteTitle}>
                {messages.visitorNote}
              </AppText>
            </View>
            <AppText style={contentTextStyle}>{place.content.visitNote}</AppText>
          </View>
        ) : null}
        {hasStory ? (
          <View
            testID="place-story-section"
            style={styles.section}
            onLayout={(event) => {
              sections.current.story = event.nativeEvent.layout.y;
              focusSection();
            }}
          >
            <AppText accessibilityRole="header" style={styles.sectionTitle}>{messages.story}</AppText>
            <AppText style={contentTextStyle}>{place.content.story}</AppText>
          </View>
        ) : null}
        <View
          testID="place-audio-section"
          onLayout={(event) => {
            sections.current.audio = event.nativeEvent.layout.y;
            focusSection();
          }}
        >
          {audio && premium.active ? (
            <NativeAudioPlayer
              key={`${identity.citySlug}:${place.slug}:${locale}`}
              source={citywalkApi.resolveUrl(audio.url)}
              title={`${nativePlaceName(
                identity.citySlug,
                place.slug,
                place.content.name,
                locale,
              )} ${messages.audioGuide}`}
              durationSeconds={audio.durationSeconds}
              assetLocale={audio.locale}
            />
          ) : audio ? (
            <View
              testID="premium-audio-locked"
              style={[styles.audioPremium, nativeTextBlock(direction)]}
            >
              <View style={[styles.audioPremiumHeading, nativeRowStyle(direction)]}>
                <View style={styles.audioPremiumIcon}>
                  <NativeIcon
                    ios="headphones"
                    android="headphones"
                    color={colors.primary}
                    size={20}
                  />
                </View>
                <View style={styles.audioPremiumCopy}>
                  <AppText variant="label">
                    {translate(locale, "cityPass.premiumLabel")}
                  </AppText>
                  <AppText variant="metadata" style={styles.muted}>
                    {premium.status === "loading"
                      ? translate(locale, "loading.account")
                      : translate(locale, "profile.passFreeDescription")}
                  </AppText>
                </View>
                <NativeIcon
                  ios="lock.fill"
                  android="lock"
                  color={colors.textMuted}
                  size={16}
                />
              </View>
              {premium.status !== "loading" ? (
                <Link href="/premium" asChild>
                  <PrimaryButton
                    compact
                    label={translate(locale, "profile.explorePass")}
                    tone="secondary"
                  />
                </Link>
              ) : null}
            </View>
          ) : (
            <View style={[styles.audioNotice, nativeTextBlock(direction)]}>
              <AppText variant="label">{messages.audioGuide}</AppText>
              <View testID="audio-unavailable-notice">
                <AppText variant="metadata" style={styles.muted}>
                  {translate(locale, "place.audioMissingDescription")}
                </AppText>
              </View>
            </View>
          )}
        </View>
        {place.content.facts?.length ? (
          <View style={styles.factsList}>
            <SectionTitle>{translate(locale, "place.quickFacts")}</SectionTitle>
            {place.content.facts.map((fact, index) => (
              <View key={`${place.slug}-fact-${index}`} style={styles.fact}>
                {fact.label ? (
                  <AppText variant="label" style={contentTextStyle}>
                    {fact.label}
                  </AppText>
                ) : null}
                <AppText style={contentTextStyle}>{fact.value}</AppText>
              </View>
            ))}
          </View>
        ) : null}
        {verifiedSources.length ? (
          <View style={styles.verifiedCard}>
            <View style={styles.verifiedHeader}>
              <View style={styles.verifiedBadge}>
                <NativeIcon
                  ios="checkmark.shield.fill"
                  android="verified"
                  color={colors.teal}
                  size={20}
                />
              </View>
              <View style={styles.verifiedCopy}>
                <AppText variant="cardTitle">{t.verified}</AppText>
                <AppText variant="metadata" style={styles.muted}>
                  {translate(locale, "place.verifiedSummary")}
                </AppText>
              </View>
            </View>
            {latestVerifiedAt ? (
              <AppText variant="metadata" style={styles.muted}>
                {t.lastChecked}: {formatVerifiedDate(locale, latestVerifiedAt)}
              </AppText>
            ) : null}
            <PressableSurface
              accessibilityLabel={translate(
                locale,
                sourcesExpanded ? "place.hideSources" : "place.viewSources",
              )}
              accessibilityRole="button"
              accessibilityState={{ expanded: sourcesExpanded }}
              onPress={() => setSourcesExpanded((current) => !current)}
              style={styles.sourcesToggle}
            >
              <AppText variant="label" style={styles.sourcesToggleText}>
                {translate(
                  locale,
                  sourcesExpanded ? "place.hideSources" : "place.viewSources",
                )}
              </AppText>
              <NativeIcon
                ios={sourcesExpanded ? "chevron.up" : "chevron.down"}
                android={
                  sourcesExpanded ? "keyboard_arrow_up" : "keyboard_arrow_down"
                }
                color={colors.textMuted}
                size={17}
              />
            </PressableSurface>
            {sourcesExpanded ? (
              <MotionView style={styles.sourceList}>
                {verifiedSources.map((source) => (
                  <PressableSurface
                    key={source.url}
                    accessibilityLabel={source.label}
                    accessibilityRole="link"
                    onPress={() => {
                      void Linking.openURL(source.url).catch(() =>
                        setSaveMessage(messages.unavailable),
                      );
                    }}
                    style={styles.sourceRow}
                  >
                    <View style={styles.sourceCopy}>
                      <AppText variant="label">{source.label}</AppText>
                      <AppText variant="metadata" style={styles.muted}>
                        {t.lastChecked}: {formatVerifiedDate(locale, source.verifiedAt)}
                      </AppText>
                    </View>
                    <NativeIcon
                      ios="arrow.up.right"
                      android="open_in_new"
                      color={colors.primary}
                      size={17}
                    />
                  </PressableSurface>
                ))}
              </MotionView>
            ) : null}
          </View>
        ) : null}
        {guideState.status === "available" && guideState.data.eligible ? (
          <Link
            href={{
              pathname: "/city/[citySlug]/guide/[placeSlug]",
              params: {
                citySlug: identity.citySlug,
                placeSlug: identity.placeSlug,
              },
            }}
            asChild
          >
            <PressableSurface
              accessibilityLabel={messages.askCitywalk}
              accessibilityRole="link"
              style={StyleSheet.flatten([
                styles.askAction,
                nativeRowStyle(direction),
              ])}
            >
              <View style={styles.askIcon}>
                <NativeIcon
                  ios="sparkles"
                  android="auto_awesome"
                  color={colors.primary}
                  size={18}
                />
              </View>
              <AppText variant="label" style={styles.askLabel}>
                {messages.askCitywalk}
              </AppText>
              <NativeIcon
                ios={direction === "rtl" ? "chevron.left" : "chevron.right"}
                android={
                  direction === "rtl" ? "chevron_left" : "chevron_right"
                }
                color={colors.textMuted}
                size={17}
              />
            </PressableSurface>
          </Link>
        ) : null}
        <View style={[styles.returnLinks, nativeRowStyle(direction)]}>
          <Link href={{ pathname: "/city/[citySlug]/explore", params: { citySlug: identity.citySlug } }} asChild><PrimaryButton wrapLabel label={t.explore} tone="secondary" /></Link>
          <Link href={{ pathname: "/city/[citySlug]", params: { citySlug: identity.citySlug } }} asChild><PrimaryButton wrapLabel label={copy.city} tone="secondary" /></Link>
        </View>
        {trip && !tripCompleted ? (
          <View style={styles.tripPanel}>
            <AppText variant="caption" style={styles.tripNextContext}>
              {nextStop ? messages.nextStop : messages.finishTrip}
            </AppText>
            <View style={styles.tripActions}>
              {previousStop ? (
                <Link
                  href={{
                    pathname: "/city/[citySlug]/place/[placeSlug]",
                    params: previousStop,
                  }}
                  replace
                  asChild
                >
                  <PrimaryButton
                    haptic="light"
                    label={messages.previousStop}
                    leadingIcon={
                      <NativeIcon
                        ios={
                          direction === "rtl" ? "arrow.right" : "arrow.left"
                        }
                        android={
                          direction === "rtl" ? "arrow_forward" : "arrow_back"
                        }
                        color={colors.primary}
                        size={18}
                      />
                    }
                    style={styles.tripAction}
                    tone="secondary"
                  />
                </Link>
              ) : null}
              {nextStop ? (
                <Link
                  href={{
                    pathname: "/city/[citySlug]/place/[placeSlug]",
                    params: nextStop,
                  }}
                  replace
                  asChild
                >
                  <PrimaryButton
                    haptic="light"
                    label={messages.nextStop}
                    style={styles.tripAction}
                    trailingIcon={
                      <NativeIcon
                        ios={
                          direction === "rtl" ? "arrow.left" : "arrow.right"
                        }
                        android={
                          direction === "rtl" ? "arrow_back" : "arrow_forward"
                        }
                        color="#FFFFFF"
                        size={18}
                      />
                    }
                  />
                </Link>
              ) : (
                <PrimaryButton
                  haptic="success"
                  label={messages.finishTrip}
                  onPress={() => setTripCompleted(true)}
                  style={styles.tripAction}
                  tone="success"
                />
              )}
            </View>
          </View>
        ) : null}
        {trip && tripCompleted ? (
          <MotionView>
            <EmptyState
              action={
                <Link
                  href={{
                    pathname: "/city/[citySlug]",
                    params: { citySlug: trip.citySlug },
                  }}
                  replace
                  asChild
                >
                  <PrimaryButton
                    label={messages.returnToCity}
                    tone="secondary"
                  />
                </Link>
              }
              description={messages.tripCompleteDescription.replace(
                "{count}",
                String(trip.stopSlugs.length),
              )}
              icon={
                <NativeIcon
                  ios="checkmark"
                  android="check"
                  color={colors.success}
                  size={30}
                />
              }
              title={messages.tripComplete}
            />
          </MotionView>
        ) : null}
      </MotionView>
    </Screen>
  );
}

const styles = StyleSheet.create({
  audioPremium: {
    backgroundColor: colors.primarySoft,
    borderRadius: radius.md,
    gap: spacing.sm,
    padding: spacing.md,
  },
  audioPremiumHeading: {
    alignItems: "center",
    flexDirection: "row",
    gap: spacing.sm,
  },
  audioPremiumIcon: {
    alignItems: "center",
    backgroundColor: colors.surface,
    borderRadius: radius.pill,
    height: 40,
    justifyContent: "center",
    width: 40,
  },
  audioPremiumCopy: { flex: 1, gap: 2, minWidth: 0 },
  audioNotice: {
    borderBottomColor: colors.border,
    borderBottomWidth: StyleSheet.hairlineWidth,
    borderTopColor: colors.border,
    borderTopWidth: StyleSheet.hairlineWidth,
    gap: 2,
    paddingVertical: spacing.sm,
  },
  content: { gap: spacing.md },
  heroFrame: { borderRadius: radius.hero, overflow: "hidden" },
  hero: {
    width: "100%",
