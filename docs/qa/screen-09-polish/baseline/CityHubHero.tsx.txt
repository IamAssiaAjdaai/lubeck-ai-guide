import { Link } from "expo-router";
import { Platform, StyleSheet, View, useWindowDimensions } from "react-native";
import type { PublicCity } from "../lib/api/contracts";
import { citywalkApi } from "../lib/api/instance";
import { selectImageUrl, selectPrimaryImageMedia } from "../lib/api/media";
import { nativeCityName } from "../lib/displayNames";
import { welcomePhotos } from "../lib/welcomeCopy";
import { useNativeLocale } from "../localization/LocaleProvider";
import { CityPhotoCredits } from "./CityPhotoCredits";
import { NativeContentImage } from "./NativeContentImage";
import { NativeIcon } from "./NativeIcon";
import { AppText, PressableSurface } from "./ui";
import lubeckPhoto from "../../assets/images/introduction-holstentor.jpg";

export function CityHubHero({ city }: { city: PublicCity }) {
  const { locale, direction } = useNativeLocale();
  const { fontScale = 1 } = useWindowDimensions();
  const name = nativeCityName(city.slug, city.content.name, locale);
  const copy = {
    en: { switch: "Change city", greeting: "Your city. Your day.", explore: "Explore" },
    de: { switch: "Stadt wechseln", greeting: "Deine Stadt. Dein Tag.", explore: "Entdecke" },
    es: { switch: "Cambiar ciudad", greeting: "Tu ciudad. Tu día.", explore: "Explora" },
    nl: { switch: "Andere stad", greeting: "Jouw stad. Jouw dag.", explore: "Ontdek" },
    da: { switch: "Skift by", greeting: "Din by. Din dag.", explore: "Udforsk" },
    sv: { switch: "Byt stad", greeting: "Din stad. Din dag.", explore: "Utforska" },
    ar: { switch: "تغيير المدينة", greeting: "مدينتك. يومك.", explore: "اكتشف" },
  }[locale];
  const media = selectPrimaryImageMedia(city.media ?? []);
  const uri = selectImageUrl(media, undefined, undefined, "hero");
  // This licensed bundled photo belongs only to Lübeck; future cities use their own published media.
  const bundled = city.slug === "lubeck";
  const source = bundled ? lubeckPhoto : uri ? { uri: citywalkApi.resolveUrl(uri) } : undefined;
  return <View testID="city-hub-hero" style={styles.hero}>
    <View style={[styles.copy, { direction }]}>
      <Link href="/?section=cities" asChild>
        <PressableSurface accessibilityRole="link" accessibilityLabel={`${name} · ${copy.switch}`} style={styles.switch}>
          <AppText style={styles.city}>{name}</AppText>
          <NativeIcon ios="chevron.down" android="expand_more" color="#FFFFFF" size={18} />
        </PressableSurface>
      </Link>
      <AppText style={styles.greeting}>{copy.greeting}</AppText>
      <AppText accessibilityRole="header" style={[styles.heading, fontScale > 1.5 && { fontSize: 26, lineHeight: 32 }, fontScale > 2.5 && { fontSize: 16, lineHeight: 22 }]}>{copy.explore} {name}</AppText>
    </View>
    {source ? <View style={styles.photoArea}>
      <NativeContentImage source={source} contentFit="cover" contentPosition="center" accessibilityLabel={name} style={styles.photo} />
      <CityPhotoCredits cityName={name} photo={bundled ? welcomePhotos.find(photo => photo.id === city.slug) : undefined} attribution={bundled ? undefined : media?.attribution} />
    </View> : null}
  </View>;
}
const styles = StyleSheet.create({
  hero: { backgroundColor: "#09234D", borderRadius: 24, overflow: "hidden" },
  copy: { paddingHorizontal: 22, paddingTop: 10, paddingBottom: 22 },
  switch: { minHeight: 48, alignSelf: "flex-start", flexDirection: "row", alignItems: "center", gap: 10, paddingVertical: 8 },
  city: { color: "#FFFFFF", fontSize: 20, lineHeight: 26, fontWeight: "700", flexShrink: 1 },
  greeting: { color: "#D6E6FA", fontSize: 14, lineHeight: 21, marginTop: 7 },
  heading: { color: "#FFFFFF", fontFamily: Platform.OS === "ios" ? "Georgia" : "serif", fontSize: 31, lineHeight: 38, fontWeight: "700", marginTop: 3 },
  photoArea: { height: 174, backgroundColor: "#155CAF" }, photo: { width: "100%", height: "100%" },
});
